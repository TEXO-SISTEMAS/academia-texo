Los datos de progreso existen pero los documentos padre no.

Necesitamos crear los documentos padre en /progress/{userId}

Crear script /scripts/create-progress-parent-docs.ts:

import { db } from '../lib/firebase'
import { collection, getDocs, setDoc, doc } from 'firebase/firestore'

async function createParentDocs() {
  console.log('Buscando usuarios con progreso...')
  
  // Buscar en todos los cursos quiénes están inscritos
  const usersSet = new Set<string>()
  
  const coursesRef = collection(db, 'courses')
  const coursesSnap = await getDocs(coursesRef)
  
  for (const courseDoc of coursesSnap.docs) {
    const courseId = courseDoc.id
    
    // Buscar en progress manualmente
    // Ya sabemos que participante@texo.com.py existe
    const knownUsers = ['participante@texo.com.py', 'dsosa@texo.com.py']
    
    for (const userId of knownUsers) {
      try {
        const coursesRef = collection(db, `progress/${userId}/courses`)
        const snap = await getDocs(coursesRef)
        if (snap.size > 0) {
          usersSet.add(userId)
        }
      } catch (e) {
        // Usuario no existe
      }
    }
  }
  
  console.log(`Usuarios encontrados: ${usersSet.size}`)
  
  for (const userId of usersSet) {
    await setDoc(doc(db, `progress/${userId}`), {
      userId,
      createdAt: new Date()
    })
    console.log(`✓ Creado documento para ${userId}`)
  }
}

createParentDocs()

Ejecutar: npx tsx scripts/create-progress-parent-docs.ts
