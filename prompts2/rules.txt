Las reglas de Firestore son demasiado permisivas y no manejan 
correctamente las subcolecciones de progreso.

Por favor creá un archivo nuevo:
firestore.rules

Con este contenido:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Usuarios - solo lectura propia
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Usuarios permitidos - solo lectura propia
    match /allowedUsers/{email} {
      allow read: if request.auth != null;
      allow write: if false; // Solo desde Admin SDK
    }
    
    // Cursos - lectura pública, escritura solo artesanos
    match /courses/{courseId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
                     get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'artesano';
      
      match /chapters/{chapterId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && 
                       get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'artesano';
        
        match /resources/{resourceId} {
          allow read: if request.auth != null;
          allow write: if request.auth != null && 
                         get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'artesano';
        }
      }
    }
    
    // Progreso - lectura y escritura propia
    match /progress/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      
      match /courses/{courseId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
        
        match /resources/{resourceId} {
          allow read, write: if request.auth != null && request.auth.uid == userId;
        }
      }
    }
    
    // Historial de login - solo escritura propia
    match /loginHistory/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Audit log - solo lectura para artesanos
    match /auditLog/{logId} {
      allow read: if request.auth != null && 
                    get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'artesano';
      allow write: if false; // Solo desde servidor
    }
  }
}

Después ejecutá:
git add firestore.rules
git commit -m "feat: add proper Firestore security rules"
git push origin main
