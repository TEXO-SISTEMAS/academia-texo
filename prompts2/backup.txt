2. FUNCIONES en lib/firestore.ts:

   - deleteCourse(courseId) → updateDoc({ deleted: true, deletedAt: timestamp })
   - deleteChapter(courseId, chapterId) → updateDoc({ deleted: true, deletedAt: timestamp })
   - deleteResource(courseId, chapterId, resourceId) → updateDoc({ deleted: true, deletedAt: timestamp })
   - deleteAcademyContent(docId) → updateDoc({ deleted: true, deletedAt: timestamp })

   - getResourcesByChapter() → where('deleted', '!=', true)
   - getAllAcademyContent() → where('deleted', '!=', true)
   - getPublishedAcademyContent() → where('deleted', '!=', true)
