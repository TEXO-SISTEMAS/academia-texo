Crear backup automático semanal que suba a Google Drive.

PASO 1 - Crear API route con Cron Job:

Crear /app/api/cron/backup/route.ts:

import { NextResponse } from 'next/server'
import { db } from '@/lib/firebase'
import { collection, getDocs } from 'firebase/firestore'
import { google } from 'googleapis'

export async function GET(request: Request) {
  // Verificar que venga de Vercel Cron
  const authHeader = request.headers.get('authorization')
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  try {
    // Exportar Firestore
    const backup = {
      timestamp: new Date().toISOString(),
      users: [],
      allowedUsers: [],
      courses: []
    }

    // Colectar data (reutilizar lógica del script existente)
    const usersSnap = await getDocs(collection(db, 'users'))
    backup.users = usersSnap.docs.map(d => ({ id: d.id, ...d.data() }))
    
    // ... resto de colecciones

    // Convertir a JSON
    const jsonData = JSON.stringify(backup, null, 2)
    const fileName = `backup-${new Date().toISOString().split('T')[0]}.json`

    // Subir a Google Drive
    const auth = new google.auth.GoogleAuth({
      credentials: JSON.parse(process.env.GOOGLE_SERVICE_ACCOUNT!),
      scopes: ['https://www.googleapis.com/auth/drive.file']
    })

    const drive = google.drive({ version: 'v3', auth })
    
    const fileMetadata = {
      name: fileName,
      parents: [process.env.BACKUP_DRIVE_FOLDER_ID!]
    }
    
    const media = {
      mimeType: 'application/json',
      body: Buffer.from(jsonData)
    }

    await drive.files.create({
      requestBody: fileMetadata,
      media: media,
      fields: 'id'
    })

    return NextResponse.json({ 
      success: true, 
      message: `Backup creado: ${fileName}` 
    })
  } catch (error) {
    console.error('Error en backup:', error)
    return NextResponse.json({ error: 'Backup failed' }, { status: 500 })
  }
}

PASO 2 - Configurar Vercel Cron:

Crear /vercel.json:

{
  "crons": [{
    "path": "/api/cron/backup",
    "schedule": "0 2 * * 0"
  }]
}

Esto ejecuta el backup cada domingo a las 2 AM.

CRON_SECRET=tu-secreto-aleatorio-seguro
