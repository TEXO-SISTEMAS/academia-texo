2. En lib/firebase-admin.ts:
"Module not found: Can't resolve '../service-account-firebase.json'"

El require del archivo local no debe estar en el código que se sube a Vercel.
Cambiar la función getServiceAccount() para que en Vercel solo use 
la variable de entorno, y el require sea solo un fallback de desarrollo:

function getServiceAccount() {
  if (process.env.FIREBASE_SERVICE_ACCOUNT_JSON) {
    try {
      return JSON.parse(process.env.FIREBASE_SERVICE_ACCOUNT_JSON);
    } catch {
      // continuar al fallback
    }
  }
  if (process.env.NODE_ENV === 'development') {
    try {
      // eslint-disable-next-line @typescript-eslint/no-require-imports
      return require('../service-account-firebase.json');
    } catch {
      // continuar
    }
  }
  throw new Error('No se pudo cargar el service account de Firebase.');
}
