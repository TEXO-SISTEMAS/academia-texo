La cookie se setea correctamente en /login pero se pierde al navegar a /bienvenida.

FIX INMEDIATO:

En /app/bienvenida/page.tsx:

1. ELIMINAR completamente la dependencia de onAuthStateChanged

2. Leer el role directamente de la cookie en el useEffect:

useEffect(() => {
  // Leer cookie directamente
  const cookies = document.cookie.split(';')
  const authCookie = cookies.find(c => c.trim().startsWith('auth='))
  
  if (!authCookie) {
    router.push('/login')
    return
  }
  
  try {
    const cookieValue = authCookie.split('=')[1]
    const decoded = JSON.parse(decodeURIComponent(cookieValue))
    const role = decoded.role
    
    console.log('[BIENVENIDA] role desde cookie:', role)
    
    if (role === 'admin') {
      router.push('/admin')
    } else if (role === 'artesano') {
      router.push('/artesano/dashboard')
    } else {
      router.push('/participante/dashboard')
    }
  } catch (error) {
    console.error('[BIENVENIDA] error leyendo cookie:', error)
    router.push('/login')
  }
}, [router])
